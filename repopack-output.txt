"""Statistics: univariate 95% confidence intervals (Student's t) and plot yield."""
import math

import numpy as np
from scipy import stats

from config import MIN_N


def mean_ci(values, confidence: float = 0.95) -> dict:
    """Mean and confidence interval of one group using Student's t-distribution."""
    data = np.asarray(values, dtype=float)
    n = len(data)
    if n < 2:
        raise ValueError("At least 2 values are needed for a confidence interval.")
    mean = float(data.mean())
    sd = float(data.std(ddof=1))                     # sample standard deviation
    sem = sd / math.sqrt(n)                          # standard error of the mean
    t_crit = float(stats.t.ppf(1 - (1 - confidence) / 2, n - 1))
    margin = t_crit * sem
    return {"n": n, "mean": mean, "sd": sd, "sem": sem, "t_value": t_crit,
            "margin": margin, "ci_lower": mean - margin, "ci_upper": mean + margin}


def _clean(values) -> list:
    """Blank (None) nitrate values are ignored and do not count towards n."""
    out = []
    for v in values:
        if v is None or (isinstance(v, str) and not v.strip()):
            continue
        out.append(float(v))
    return [v for v in out if not math.isnan(v)]


def compare_nitrate(control_values, variant_values, min_n: int = MIN_N) -> dict:
    """Control vs variant nitrate comparison.

    The chart is only produced when BOTH groups have at least min_n readings.
    """
    control, variant = _clean(control_values), _clean(variant_values)
    result = {"n_control": len(control), "n_variant": len(variant), "min_n": min_n}
    if len(control) < min_n or len(variant) < min_n:
        result.update(sufficient=False, notice=(
            f"More records needed: at least {min_n} nitrate readings per treatment group "
            f"are required for a 95% confidence interval "
            f"(control has {len(control)}, variant has {len(variant)})."))
        return result

    c, v = mean_ci(control), mean_ci(variant)
    overlap = not (c["ci_upper"] < v["ci_lower"] or v["ci_upper"] < c["ci_lower"])
    difference = v["mean"] - c["mean"]
    result.update(
        sufficient=True, control=c, variant=v, overlap=overlap,
        difference=difference,
        difference_pct=(difference / c["mean"] * 100) if c["mean"] else None,
        # Green: intervals do not overlap (clear difference). Yellow: they overlap.
        band="yellow" if overlap else "green",
        band_text=("The 95% intervals overlap: no clear difference yet."
                   if overlap else "The 95% intervals do not overlap: clear difference."))
    return result


def plot_yield(harvest_records: list, area_m2) -> float:
    """Yield in g/m2 = total fresh weight of the plot's harvest records / plot area.

    Returns None while no harvest has been marked complete ("Harvest in progress").
    """
    if not area_m2 or not any(r.get("is_harvest_complete") for r in harvest_records):
        return None
    total = sum(float(r["fresh_weight_g"]) for r in harvest_records
                if r.get("fresh_weight_g") is not None)
    return total / float(area_m2)


def compare_all_groups(control_values, variant_values_by_group: dict, min_n: int = MIN_N) -> list:
    """Compare the control with EVERY treatment group (TGID-1 to TGID-5).

    Returns one compare_nitrate() result per group, each with a "group" key added.
    """
    results = []
    for group, values in variant_values_by_group.items():
        result = compare_nitrate(control_values, values, min_n)
        result["group"] = group
        results.append(result)
    return results


def summarise(values) -> dict:
    """n, mean and sample SD of the non-blank values (no confidence interval)."""
    data = _clean(values)
    n = len(data)
    return {"n": n,
            "mean": float(np.mean(data)) if n else None,
            "sd": float(np.std(data, ddof=1)) if n > 1 else None}


def verdict(result: dict) -> str:
    """Plain-words result of one control-vs-group comparison."""
    if not result["sufficient"]:
        return "Insufficient data"
    if result["overlap"]:
        return "No clear difference"
    return "Clear difference (higher)" if result["difference"] > 0 else "Clear difference (lower)"

"""Thin wrapper the Streamlit app uses to call the FastAPI engine."""
import requests

import config


def _headers(role: str) -> dict:
    return {"X-User-Role": role}


def signup(username, password, full_name, contact):
    """Returns (ok, user_dict_or_error_text)."""
    try:
        r = requests.post(f"{config.API_URL}/auth/signup", timeout=8, json={
            "username": username, "password": password,
            "full_name": full_name, "contact": contact})
    except requests.RequestException:
        return False, "Cannot reach the server. Sign-up needs an internet connection."
    return (True, r.json()) if r.status_code == 200 else (False, r.json().get("detail", "Sign-up failed."))


def login(username, password):
    """Online login (first time on this device). Returns (ok, user_dict_or_error_text)."""
    try:
        r = requests.post(f"{config.API_URL}/auth/login", timeout=8,
                          json={"username": username, "password": password})
    except requests.RequestException:
        return False, "Cannot reach the server."
    return (True, r.json()) if r.status_code == 200 else (False, r.json().get("detail", "Login failed."))


def get_json(path: str, role: str, **params):
    r = requests.get(f"{config.API_URL}{path}", headers=_headers(role), params=params, timeout=10)
    r.raise_for_status()
    return r.json()


def get_bytes(path: str, role: str, **params) -> bytes:
    r = requests.get(f"{config.API_URL}{path}", headers=_headers(role), params=params, timeout=20)
    r.raise_for_status()
    return r.content

def require_roles(*roles):
    """Dependency: allow the request only for the listed roles."""
    def check(x_user_role: str = Header(default="")):
        if x_user_role not in roles:
            raise HTTPException(status_code=403, detail="This role is not allowed to do that.")
        return x_user_role
    return check

@app.post("/auth/signup", tags=["auth"])
def signup(body: dict = Body(...)):
    """Create a farmer account (needs internet the first time). Returns the new OID."""
    username = str(body.get("username", "")).strip()
    password = str(body.get("password", ""))
    if len(username) < 3 or len(password) < 6:
        raise HTTPException(400, "Username needs 3+ characters and password 6+ characters.")
    store = get_store()
    if store.get_user(username):
        raise HTTPException(409, "That username is already taken.")
    salt, pw_hash = hash_password(password)
    # Self sign-up is always a Small Farmer. Stakeholder accounts are created by an admin.
    oid = store.create_user(username, config.ROLE_FARMER, body.get("full_name"),
                           body.get("contact"), salt, pw_hash)
    return {"oid": oid, "username": username, "role": config.ROLE_FARMER,
            "full_name": body.get("full_name")}


@app.post("/auth/login", tags=["auth"])
def login(body: dict = Body(...)):
    user = get_store().get_user(str(body.get("username", "")).strip())
    if not user or not verify_password(str(body.get("password", "")), user["salt"],
                                       user["password_hash"]):
        raise HTTPException(401, "Wrong username or password.")
    return {"oid": user["oid"], "username": user["username"], "role": user["role"],
            "full_name": user["full_name"]}

@app.post("/api/v1/trials", tags=["trials"])
def upsert_trial(rec: dict = Body(...), _=Depends(farmer_only)):
    """Sync endpoint. Validates with the same rules as the phone, then saves to the cloud.

    200 {"status":"SUCCESS","cloud_record_id":"EP-T101"}  -> phone marks SYNCED
    422 {"status":"REJECTED","error":"..."}               -> phone marks FLAGGED_FOR_REVIEW
    5xx / timeout                                         -> record stays PENDING (retried)
    """
    errors = validate_record(rec)
    if errors:
        return JSONResponse(status_code=422,
                            content={"status": "REJECTED", "error": format_errors(errors)})
    try:
        record_id = get_store().upsert_trial(rec)
    except Exception as exc:  # cloud problem: not the farmer's fault, keep PENDING
        raise HTTPException(503, f"Cloud unavailable: {exc}")
    return {"status": "SUCCESS", "cloud_record_id": record_id}

def _check_variant(variant_group_id: str):
    if variant_group_id not in config.VARIANT_GROUPS:
        raise HTTPException(400, "variant_group_id must be TGID-1 to TGID-5.")


@app.get("/api/v1/analytics/nitrate-ci", tags=["analytics"])
def nitrate_ci(site_id: str = Query(...), crop_id: str = Query(...),
               variant_group_id: str = Query(...), trial_week: str = Query(...),
               _=Depends(stakeholder)):
    """Mean +/- 95% CI (Student's t) of nitrate for control vs variant, only when n >= 5 in both."""
    _check_variant(variant_group_id)
    store = get_store()

    def nitrate(group):
        rows = store.list_trials(site_id=site_id, crop_id=crop_id,
                                 treatment_group_id=group, trial_week=trial_week)
        return [r["nitrate_mg_l"] for r in rows]

    result = analytics.compare_nitrate(nitrate("TGID-0"), nitrate(variant_group_id))
    result.update(site_id=site_id, crop_id=crop_id, trial_week=trial_week,
                  variant_group_id=variant_group_id)
    return result


@app.get("/api/v1/analytics/plot-yield", tags=["analytics"])
def plot_yield(site_id: str = Query(...), crop_id: str = Query(...),
               treatment_group_id: str = Query(...), _=Depends(stakeholder)):
    """Yield (g/m2) for one plot. 'Harvest in progress' until a final harvest is confirmed."""
    plot_id = f"{site_id}-{crop_id}-{treatment_group_id}"
    store = get_store()
    plot = store.get_plot(plot_id)
    rows = store.list_trials(plot_id=plot_id, visit_type="HARVEST_VISIT")
    value = analytics.plot_yield(rows, plot["area_m2"] if plot else None)
    return {"plot_id": plot_id, "yield_g_m2": value,
            "status": "Harvest in progress" if value is None else "Complete"}

@app.get("/api/v1/dossier", tags=["reports"])
def dossier(site_id: str = Query(...), crop_id: str = Query(...),
            variant_group_id: str = Query(...), _=Depends(dalrrd_only)):
    """DALRRD Act 36 dossier: ZIP with anonymised CSV and the combined protocol text."""
    _check_variant(variant_group_id)
    store = get_store()
    records = (store.list_trials(site_id=site_id, crop_id=crop_id, treatment_group_id="TGID-0")
               + store.list_trials(site_id=site_id, crop_id=crop_id,
                                   treatment_group_id=variant_group_id))
    data = build_dossier(records, store.get_protocol("TGID-0"),
                         store.get_protocol(variant_group_id), variant_group_id)
    filename = f"Act36_Dossier_{site_id}_{crop_id}_{variant_group_id}.zip"
    return Response(content=data, media_type="application/zip",
                    headers={"Content-Disposition": f'attachment; filename="{filename}"'})

def status_banner():
    """System Status: Online (Synced) / Offline (X records queued)."""
    pending = db.count_by_status()["PENDING"]
    flagged = db.count_by_status()["FLAGGED_FOR_REVIEW"]
    if sync_worker.is_online():
        if pending:
            st.info(f"System Status: Online ({pending} record(s) waiting to sync)")
        else:
            st.success("System Status: Online (Synced)")
    else:
        st.warning(f"System Status: Offline ({pending} Records Queued)")
    if flagged:
        st.error(f"{flagged} record(s) were rejected by the cloud. Open 'My Records' to see why.")


def do_login(username: str, password: str):
    """Offline-first login: saved credentials first, then the server (first time only)."""
    user = db.verify_local_user(username, password)
    if user:
        return user, None
    if not sync_worker.is_online():
        return None, ("No saved login for that user on this device. "
                      "Connect to the internet to log in the first time.")
    ok, result = api_client.login(username, password)
    if not ok:
        return None, result
    db.cache_user(result["oid"], result["username"], result["role"], result["full_name"], password)
    try:  # first login on this device: pull site / plot setup data
        sync_worker.sync_setup_data()
    except Exception:
        pass
    return result, None

"""Dashboard charts (Plotly). The app calls these; the maths stays in analytics.py."""
import plotly.graph_objects as go

GREEN, YELLOW, GREY, DARK = "#2e7d32", "#f9a825", "#5b6b7b", "#1f2933"


def ci_figure(control: dict, results: list):
    """Mean +/- 95% CI for the control and every treatment group that has enough data.

    control : the "control" dict of any sufficient comparison (mean, ci_lower, ci_upper, margin)
    results : the list of compare_nitrate() results, one per treatment group, each with "group"
    Green = the group's interval does not overlap the control's; yellow = it overlaps.
    """
    names = ["Control (TGID-0)"]
    means, margins, colours = [control["mean"]], [control["margin"]], [GREY]
    for r in results:
        if r["sufficient"]:
            names.append(r["group"])
            means.append(r["variant"]["mean"])
            margins.append(r["variant"]["margin"])
            colours.append(GREEN if r["band"] == "green" else YELLOW)

    fig = go.Figure()
    # the control's own 95% interval as a grey band across the whole chart
    fig.add_hrect(y0=control["ci_lower"], y1=control["ci_upper"], fillcolor=GREY,
                  opacity=0.15, line_width=0)
    fig.add_hline(y=control["mean"], line_dash="dash", line_color=GREY,
                  annotation_text="Control mean", annotation_position="top left")
    fig.add_trace(go.Scatter(
        x=names, y=means, mode="markers",
        marker=dict(size=14, color=colours, line=dict(width=1, color=DARK)),
        error_y=dict(type="data", array=margins, visible=True, thickness=2, width=10, color=DARK),
        hovertemplate="%{x}<br>mean %{y:.1f} mg/L<extra></extra>"))
    fig.update_xaxes(categoryorder="array", categoryarray=names)
    fig.update_layout(
        yaxis_title="Nitrate (mg/L), mean and 95% CI", height=400, showlegend=False,
        margin=dict(l=10, r=10, t=10, b=10),
        hoverlabel=dict(bgcolor=DARK, font=dict(color="#ffffff", size=14)))
    return fig


def yield_figure(yields: dict):
    """Yield (g/m2) per treatment group. A group still being harvested shows 'in progress'."""
    groups = list(yields)
    fig = go.Figure(go.Bar(
        x=groups,
        y=[v if v is not None else 0 for v in yields.values()],
        marker_color=[GREY if g == "TGID-0" else GREEN for g in groups],
        text=["in progress" if v is None else f"{v:.0f}" for v in yields.values()],
        textposition="outside",
        hovertemplate="%{x}<br>%{y:.0f} g/m2<extra></extra>"))
    fig.update_layout(yaxis_title="Yield (g/m2)", height=320, showlegend=False,
                      margin=dict(l=10, r=10, t=20, b=10))
    return fig

"""Cloud storage behind the API.

Two interchangeable back ends with the SAME methods:

* MockCloudStore  - a local SQLite file that behaves like the cloud. Used when
                    no Supabase keys are configured (development, tests, demo).
* SupabaseStore   - the real Supabase PostgreSQL database.

get_store() picks Supabase when SUPABASE_URL and SUPABASE_SERVICE_KEY are set
in the environment (.env), otherwise the mock. The rest of the system never
needs to know which one it is talking to.
"""
import os
import sqlite3

import config
from database import PAYLOAD_COLUMNS

# Column mapping: the phone's local_id is the cloud's client_uuid.
CLOUD_COLUMNS = [c for c in PAYLOAD_COLUMNS if c != "local_id"]


def _to_bool(value) -> bool:
    return bool(value) and value not in ("0", "false", "False")


def _normalise_trial(row: dict) -> dict:
    """Common shape returned by both stores."""
    row = dict(row)
    row["is_harvest_complete"] = _to_bool(row.get("is_harvest_complete"))
    return row


# =========================================================== local simulator ==
class MockCloudStore:
    """SQLite file that imitates the Supabase tables (same names and behaviour)."""

    name = "local-simulator"

    def __init__(self, path: str):
        self.path = path
        self._init()

    def _conn(self):
        conn = sqlite3.connect(self.path)
        conn.row_factory = sqlite3.Row
        return conn

    def _init(self):
        cols = ", ".join(CLOUD_COLUMNS)  # typeless columns are fine in SQLite
        with self._conn() as c:
            c.executescript(f"""
                CREATE TABLE IF NOT EXISTS sites (site_id TEXT PRIMARY KEY, start_date TEXT NOT NULL);
                CREATE TABLE IF NOT EXISTS plots (plot_id TEXT PRIMARY KEY, site_id TEXT, crop_id TEXT,
                    treatment_group_id TEXT, area_m2 REAL);
                CREATE TABLE IF NOT EXISTS protocols (tgid TEXT PRIMARY KEY, markdown_text TEXT);
                CREATE TABLE IF NOT EXISTS app_users (oid TEXT PRIMARY KEY, username TEXT UNIQUE,
                    role TEXT, full_name TEXT, contact TEXT, salt TEXT, password_hash TEXT);
                CREATE TABLE IF NOT EXISTS trial_measurements (
                    seq INTEGER PRIMARY KEY AUTOINCREMENT,
                    record_id TEXT UNIQUE,
                    client_uuid TEXT UNIQUE NOT NULL,
                    {cols});
            """)

    def reset(self):
        with self._conn() as c:
            for table in ("trial_measurements", "sites", "plots", "protocols", "app_users"):
                c.execute(f"DELETE FROM {table}")
            c.execute("DELETE FROM sqlite_sequence")  # EP-T101 again on the next record

    def ping(self) -> bool:
        return True

    # ---- setup data ("Server Wins": the first one to arrive sets the baseline)
    def get_site(self, site_id):
        with self._conn() as c:
            r = c.execute("SELECT * FROM sites WHERE site_id = ?", (site_id,)).fetchone()
        return dict(r) if r else None

    def create_site_if_absent(self, site_id, start_date):
        if not self.get_site(site_id):
            with self._conn() as c:
                c.execute("INSERT INTO sites VALUES (?, ?)", (site_id, start_date))
        return self.get_site(site_id)

    def list_sites(self):
        with self._conn() as c:
            return [dict(r) for r in c.execute("SELECT * FROM sites")]

    def get_plot(self, plot_id):
        with self._conn() as c:
            r = c.execute("SELECT * FROM plots WHERE plot_id = ?", (plot_id,)).fetchone()
        return dict(r) if r else None

    def create_plot_if_absent(self, plot):
        if not self.get_plot(plot["plot_id"]):
            with self._conn() as c:
                c.execute("INSERT INTO plots VALUES (?, ?, ?, ?, ?)",
                          (plot["plot_id"], plot["site_id"], plot["crop_id"],
                           plot["treatment_group_id"], plot["area_m2"]))
        return self.get_plot(plot["plot_id"])

    def list_plots(self):
        with self._conn() as c:
            return [dict(r) for r in c.execute("SELECT * FROM plots")]

    def get_protocol(self, tgid):
        with self._conn() as c:
            r = c.execute("SELECT markdown_text FROM protocols WHERE tgid = ?", (tgid,)).fetchone()
        return r["markdown_text"] if r else None

    def set_protocol(self, tgid, text):
        with self._conn() as c:
            c.execute("INSERT OR REPLACE INTO protocols VALUES (?, ?)", (tgid, text))

    # ---- users
    def get_user(self, username):
        with self._conn() as c:
            r = c.execute("SELECT * FROM app_users WHERE username = ?", (username,)).fetchone()
        return dict(r) if r else None

    def next_oid(self):
        """Farmers get OID-101, OID-102, ... (900+ is reserved for stakeholders)."""
        with self._conn() as c:
            rows = [r["oid"] for r in c.execute("SELECT oid FROM app_users")]
        numbers = [int(o[4:]) for o in rows if 100 <= int(o[4:]) < 900]
        return f"OID-{(max(numbers) + 1) if numbers else 101}"

    def create_user(self, username, role, full_name, contact, salt, password_hash, oid=None):
        oid = oid or self.next_oid()
        with self._conn() as c:
            c.execute("INSERT INTO app_users VALUES (?, ?, ?, ?, ?, ?, ?)",
                      (oid, username, role, full_name, contact, salt, password_hash))
        return oid

    # ---- trial records
    def upsert_trial(self, rec) -> str:
        """Insert or update by client_uuid. Returns the cloud Record ID (EP-T###)."""
        uuid_ = rec["local_id"]
        row = {c: rec.get(c) for c in CLOUD_COLUMNS}
        row["is_harvest_complete"] = 1 if rec.get("is_harvest_complete") else 0
        with self._conn() as c:
            existing = c.execute("SELECT record_id FROM trial_measurements WHERE client_uuid = ?",
                                 (uuid_,)).fetchone()
            if existing:  # an edit: keep the permanent record_id
                sets = ", ".join(f"{k} = ?" for k in row)
                c.execute(f"UPDATE trial_measurements SET {sets} WHERE client_uuid = ?",
                          [*row.values(), uuid_])
                return existing["record_id"]
            names = ", ".join(["client_uuid", *row])
            marks = ", ".join("?" for _ in range(len(row) + 1))
            cur = c.execute(f"INSERT INTO trial_measurements ({names}) VALUES ({marks})",
                            [uuid_, *row.values()])
            record_id = f"EP-T{100 + cur.lastrowid}"
            c.execute("UPDATE trial_measurements SET record_id = ? WHERE seq = ?",
                      (record_id, cur.lastrowid))
            return record_id

    def list_trials(self, **filters):
        sql, args = "SELECT * FROM trial_measurements WHERE 1=1", []
        for key, value in filters.items():
            if value is not None:
                sql += f" AND {key} = ?"
                args.append(value)
        sql += " ORDER BY seq"
        with self._conn() as c:
            return [_normalise_trial(r) for r in c.execute(sql, args)]

class SupabaseStore:
    """Real Supabase back end. Same methods as MockCloudStore."""

    name = "supabase"

    def __init__(self, url: str, key: str):
        from supabase import create_client  # imported here so the mock needs no keys
        self.sb = create_client(url, key)

    def _rows(self, table, **eq):
        query = self.sb.table(table).select("*")
        for k, v in eq.items():
            if v is not None:
                query = query.eq(k, v)
        return query.execute().data

    def ping(self) -> bool:
        """False when the project is paused (free-tier 7-day inactivity) or unreachable."""
        try:
            self.sb.table("sites").select("site_id").limit(1).execute()
            return True
        except Exception:
            return False

    def reset(self):
        for table, col in (("trial_measurements", "client_uuid"), ("plots", "plot_id"),
                           ("sites", "site_id"), ("protocols", "tgid"), ("app_users", "oid")):
            self.sb.table(table).delete().neq(col, "").execute()

    def get_site(self, site_id):
        rows = self._rows("sites", site_id=site_id)
        return rows[0] if rows else None

    def create_site_if_absent(self, site_id, start_date):
        if not self.get_site(site_id):
            self.sb.table("sites").insert({"site_id": site_id, "start_date": start_date}).execute()
        return self.get_site(site_id)

    def list_sites(self):
        return self._rows("sites")

    def get_plot(self, plot_id):
        rows = self._rows("plots", plot_id=plot_id)
        return rows[0] if rows else None

    def create_plot_if_absent(self, plot):
        if not self.get_plot(plot["plot_id"]):
            self.sb.table("plots").insert(plot).execute()
        return self.get_plot(plot["plot_id"])

    def list_plots(self):
        return self._rows("plots")

    def get_protocol(self, tgid):
        rows = self._rows("protocols", tgid=tgid)
        return rows[0]["markdown_text"] if rows else None

    def set_protocol(self, tgid, text):
        self.sb.table("protocols").upsert({"tgid": tgid, "markdown_text": text}).execute()

    def get_user(self, username):
        rows = self._rows("app_users", username=username)
        return rows[0] if rows else None

    def next_oid(self):
        rows = [r["oid"] for r in self._rows("app_users")]
        numbers = [int(o[4:]) for o in rows if 100 <= int(o[4:]) < 900]
        return f"OID-{(max(numbers) + 1) if numbers else 101}"

    def create_user(self, username, role, full_name, contact, salt, password_hash, oid=None):
        oid = oid or self.next_oid()
        self.sb.table("app_users").insert({
            "oid": oid, "username": username, "role": role, "full_name": full_name,
            "contact": contact, "salt": salt, "password_hash": password_hash}).execute()
        return oid

    def upsert_trial(self, rec) -> str:
        row = {c: rec.get(c) for c in CLOUD_COLUMNS}
        row["client_uuid"] = rec["local_id"]
        row["is_harvest_complete"] = bool(rec.get("is_harvest_complete"))
        self.sb.table("trial_measurements").upsert(row, on_conflict="client_uuid").execute()
        saved = self._rows("trial_measurements", client_uuid=rec["local_id"])
        return saved[0]["record_id"]

    def list_trials(self, **filters):
        return [_normalise_trial(r) for r in self._rows("trial_measurements", **filters)]

    global _store

"""Local (on-phone) SQLite storage: records, setup data, offline logins, settings.

Every function takes an optional db_path so tests can use a temporary database.
"""
import sqlite3
import uuid
from datetime import date, datetime
from pathlib import Path

import config
from security import hash_password, verify_password

SCHEMA_FILE = Path(__file__).resolve().parent / "sql" / "schema_sqlite.sql"

# Every column of trial_measurements, in schema order.
RECORD_COLUMNS = [
    "local_id", "cloud_record_id", "created_timestamp", "entry_date", "site_id",
    "environment_id", "crop_id", "treatment_group_id", "plot_id", "trial_week",
    "visit_type", "observer_id", "plant_height_cm", "leaf_count",
    "fresh_weight_g", "dry_weight_g", "spad_index", "root_length_cm",
    "ph_level", "ec_level", "dissolved_oxygen", "temperature_c", "humidity_pct",
    "nitrate_mg_l", "phosphate_mg_l", "notes",
    "sync_status", "rejection_reason", "is_harvest_complete",
    "last_edited_by", "last_edited_timestamp",
]

# What is sent to the cloud: everything except the phone-only sync fields.
PAYLOAD_COLUMNS = [c for c in RECORD_COLUMNS
                   if c not in ("cloud_record_id", "sync_status", "rejection_reason")]


def connect(db_path: str = None) -> sqlite3.Connection:
    """Open a connection (rows behave like dicts)."""
    conn = sqlite3.connect(db_path or config.DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_sqlite_db(db_path: str = None) -> None:
    """Create all local tables if they do not exist yet."""
    with connect(db_path) as conn:
        conn.executescript(SCHEMA_FILE.read_text(encoding="utf-8"))


# ---------------------------------------------------------------- settings --
def get_setting(key: str, default: str = None, db_path: str = None):
    with connect(db_path) as conn:
        row = conn.execute("SELECT value FROM app_settings WHERE key = ?", (key,)).fetchone()
    return row["value"] if row else default


def set_setting(key: str, value: str, db_path: str = None) -> None:
    with connect(db_path) as conn:
        conn.execute("INSERT INTO app_settings (key, value) VALUES (?, ?) "
                     "ON CONFLICT(key) DO UPDATE SET value = excluded.value", (key, value))


def is_offline_mode(db_path: str = None) -> bool:
    """The demo 'Offline mode' switch (forces the app to behave as if offline)."""
    return get_setting("offline_mode", "0", db_path) == "1"


# ------------------------------------------------------------------- users --
def cache_user(oid: str, username: str, role: str, full_name: str, password: str,
               db_path: str = None) -> None:
    """Store a salted hash locally so this user can log in offline later."""
    salt, pw_hash = hash_password(password)
    with connect(db_path) as conn:
        conn.execute(
            "INSERT INTO users (oid, username, role, full_name, salt, password_hash) "
            "VALUES (?, ?, ?, ?, ?, ?) ON CONFLICT(oid) DO UPDATE SET "
            "username = excluded.username, role = excluded.role, full_name = excluded.full_name, "
            "salt = excluded.salt, password_hash = excluded.password_hash",
            (oid, username, role, full_name, salt, pw_hash))


def verify_local_user(username: str, password: str, db_path: str = None):
    """Offline login. Returns the user dict, or None if unknown / wrong password."""
    with connect(db_path) as conn:
        row = conn.execute("SELECT * FROM users WHERE username = ?", (username,)).fetchone()
    if row and verify_password(password, row["salt"], row["password_hash"]):
        return {"oid": row["oid"], "username": row["username"],
                "role": row["role"], "full_name": row["full_name"]}
    return None


# ----------------------------------------------------- site / plot setup ----
def plot_id_for(site_id: str, crop_id: str, treatment_group_id: str) -> str:
    """A plot is one site + one crop + one treatment group."""
    return f"{site_id}-{crop_id}-{treatment_group_id}"


def upsert_site(site_id: str, start_date: str, db_path: str = None) -> None:
    with connect(db_path) as conn:
        conn.execute("INSERT INTO sites (site_id, start_date) VALUES (?, ?) "
                     "ON CONFLICT(site_id) DO UPDATE SET start_date = excluded.start_date",
                     (site_id, start_date))


def get_site(site_id: str, db_path: str = None):
    with connect(db_path) as conn:
        row = conn.execute("SELECT * FROM sites WHERE site_id = ?", (site_id,)).fetchone()
    return dict(row) if row else None


def list_sites(db_path: str = None) -> list:
    with connect(db_path) as conn:
        return [dict(r) for r in conn.execute("SELECT * FROM sites")]


def upsert_plot(plot_id: str, site_id: str, crop_id: str, treatment_group_id: str,
                area_m2: float, db_path: str = None) -> None:
    with connect(db_path) as conn:
        conn.execute(
            "INSERT INTO plots (plot_id, site_id, crop_id, treatment_group_id, area_m2) "
            "VALUES (?, ?, ?, ?, ?) ON CONFLICT(plot_id) DO UPDATE SET area_m2 = excluded.area_m2",
            (plot_id, site_id, crop_id, treatment_group_id, area_m2))


def get_plot(plot_id: str, db_path: str = None):
    with connect(db_path) as conn:
        row = conn.execute("SELECT * FROM plots WHERE plot_id = ?", (plot_id,)).fetchone()
    return dict(row) if row else None


def list_plots(db_path: str = None) -> list:
    with connect(db_path) as conn:
        return [dict(r) for r in conn.execute("SELECT * FROM plots")]


def calculate_trial_week(start_date: str, entry_date: str):
    """Days 0-6 after the start date are TW-1, days 7-13 are TW-2, and so on.

    Returns None if the entry date is before the site start date.
    """
    days = (date.fromisoformat(entry_date) - date.fromisoformat(start_date)).days
    if days < 0:
        return None
    return f"TW-{days // 7 + 1}"


# ----------------------------------------------------------------- records --
def new_local_id() -> str:
    return str(uuid.uuid4())


def now_timestamp() -> str:
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def save_local_record(rec: dict, db_path: str = None) -> dict:
    """Save (or re-save after an edit) a record as PENDING. Works with no internet."""
    rec = dict(rec)
    rec.setdefault("local_id", new_local_id())
    rec.setdefault("created_timestamp", now_timestamp())
    rec["sync_status"] = "PENDING"
    rec["rejection_reason"] = None
    rec["is_harvest_complete"] = 1 if rec.get("is_harvest_complete") else 0
    values = [rec.get(c) for c in RECORD_COLUMNS]
    placeholders = ", ".join("?" for _ in RECORD_COLUMNS)
    with connect(db_path) as conn:
        conn.execute(f"INSERT OR REPLACE INTO trial_measurements "
                     f"({', '.join(RECORD_COLUMNS)}) VALUES ({placeholders})", values)
    return rec


def list_records(db_path: str = None, observer_id: str = None, status: str = None,
                 oldest_first: bool = False) -> list:
    sql, args = "SELECT * FROM trial_measurements WHERE 1=1", []
    if observer_id:
        sql += " AND observer_id = ?"
        args.append(observer_id)
    if status:
        sql += " AND sync_status = ?"
        args.append(status)
    sql += " ORDER BY created_timestamp " + ("ASC" if oldest_first else "DESC")
    with connect(db_path) as conn:
        return [dict(r) for r in conn.execute(sql, args)]


def count_by_status(db_path: str = None) -> dict:
    counts = {"PENDING": 0, "SYNCED": 0, "FLAGGED_FOR_REVIEW": 0}
    with connect(db_path) as conn:
        for row in conn.execute("SELECT sync_status, COUNT(*) AS n FROM trial_measurements "
                                "GROUP BY sync_status"):
            counts[row["sync_status"]] = row["n"]
    return counts


def mark_synced(local_id: str, cloud_record_id: str, db_path: str = None) -> None:
    with connect(db_path) as conn:
        conn.execute("UPDATE trial_measurements SET sync_status = 'SYNCED', "
                     "cloud_record_id = ?, rejection_reason = NULL WHERE local_id = ?",
                     (cloud_record_id, local_id))


def mark_flagged(local_id: str, reason: str, db_path: str = None) -> None:
    with connect(db_path) as conn:
        conn.execute("UPDATE trial_measurements SET sync_status = 'FLAGGED_FOR_REVIEW', "
                     "rejection_reason = ? WHERE local_id = ?", (reason, local_id))


def count_harvest_missing_fresh(plot_id: str, exclude_local_id: str = None,
                                db_path: str = None) -> int:
    """Stored harvest records for a plot that have no fresh weight (completeness warning)."""
    sql = ("SELECT COUNT(*) AS n FROM trial_measurements WHERE plot_id = ? "
           "AND visit_type = 'HARVEST_VISIT' AND fresh_weight_g IS NULL")
    args = [plot_id]
    if exclude_local_id:
        sql += " AND local_id != ?"
        args.append(exclude_local_id)
    with connect(db_path) as conn:
        return conn.execute(sql, args).fetchone()["n"]

"""DALRRD Act 36 dossier: a ZIP with anonymised CSV data + Markdown protocol text."""
import io
import zipfile

import pandas as pd

# Only these columns are exported. Farmer names and contact details live in the
# auth table and are never in trial rows. Free-text notes are left out because a
# farmer could type a name into them. GPS is not collected at all.
EXPORT_COLUMNS = [
    "record_id", "entry_date", "site_id", "environment_id", "crop_id",
    "treatment_group_id", "plot_id", "trial_week", "visit_type", "observer_id",
    "plant_height_cm", "leaf_count", "fresh_weight_g", "dry_weight_g", "spad_index",
    "root_length_cm", "ph_level", "ec_level", "dissolved_oxygen", "temperature_c",
    "humidity_pct", "nitrate_mg_l", "phosphate_mg_l", "is_harvest_complete",
]


def anonymise(records: list) -> pd.DataFrame:
    """Keep only the whitelisted columns (Observer ID kept for audit traceability)."""
    frame = pd.DataFrame(records)
    for col in EXPORT_COLUMNS:
        if col not in frame:
            frame[col] = None
    return frame[EXPORT_COLUMNS]


def combine_protocols(control_text, variant_text, variant_id: str) -> str:
    """Control protocol (TGID-0) followed by the selected variant's protocol."""
    return (
        "# Treatment Protocol Description\n\n"
        "## Control group (TGID-0)\n\n"
        f"{control_text or '_Protocol text not yet confirmed by the Eduponics research team._'}\n\n"
        f"## Variant group ({variant_id})\n\n"
        f"{variant_text or '_Protocol text not yet confirmed by the Eduponics research team._'}\n")


def build_dossier(records: list, control_protocol, variant_protocol, variant_id: str) -> bytes:
    """Return the dossier ZIP as bytes."""
    csv_text = anonymise(records).round(3).to_csv(index=False)
    protocol_md = combine_protocols(control_protocol, variant_protocol, variant_id)
    buffer = io.BytesIO()
    with zipfile.ZipFile(buffer, "w", zipfile.ZIP_DEFLATED) as z:
        z.writestr("trial_observation_data.csv", csv_text)
        z.writestr("treatment_protocol_description.md", protocol_md)
    return buffer.getvalue()

# AgriTrial MVP (IFS140 Group 1)

Offline-first trial data capture for small farmers, with 95% confidence-interval
analytics and a DALRRD Act 36 dossier export.

**How it works:** the farmer's app saves every record on the phone first (SQLite,
status `PENDING`). When online, the sync worker sends records through the FastAPI
engine, which validates them and writes them to Supabase. Records come back
`SYNCED` (with a cloud ID like `EP-T101`) or `FLAGGED_FOR_REVIEW` (with a reason).
Stakeholders open a dashboard that shows nitrate mean ± 95% CI (Student's t) for
control vs a treatment variant, but only when both groups have at least 5 readings.

```
Streamlit app ──► SQLite (phone) ──► sync_worker.py ──► FastAPI (api_engine.py) ──► Supabase
 (app.py)         PENDING queue        oldest first        validates, assigns IDs     (or local simulator)
                                                                  │
                                       Dashboard ◄── analytics.py ┘──► dossier.py (ZIP)
```

## 8. Known limitations (MVP scope)

- The "phone" app is a responsive web app (Streamlit), not a native Android/iOS app.
- On-device database encryption is designed (Milestone 1, Section 9) but **not built**.
- Language choice is saved in the profile; translated screens are not built yet.
- Editing a flagged record, the weather alert, voice notes and the chatbot are not built (post-MVP).
- Supabase mode needs a real project and keys; the default demo uses the local simulator.
- GPS is not collected, so there is nothing to blur or strip.

"""Act 36 PDF trial report: EVERY treatment group compared with the control.

build_pdf_report() reads the trial data from the cloud store and turns it into a PDF for
the regulator: summary of findings, a nitrate chart with 95% confidence intervals for all

# ------------------------------------------------------------------ data ----
def _week_no(week: str) -> int:
    return int(week[3:])


def collect_report_data(store, site_id: str, crop_id: str, trial_week: str = None) -> dict:
    """Gather everything the report needs from the cloud store."""
    site = store.get_site(site_id)
    trials = {g: store.list_trials(site_id=site_id, crop_id=crop_id, treatment_group_id=g)
              for g in config.TREATMENT_GROUPS}
    weeks = sorted({r["trial_week"] for rows in trials.values() for r in rows}, key=_week_no)

    def nitrate(group, week):
        return [r.get("nitrate_mg_l") for r in trials[group] if r["trial_week"] == week]

    if trial_week not in weeks:        # default: the latest week where the control has enough data
        trial_week = None
        for week in reversed(weeks):
            if analytics.summarise(nitrate("TGID-0", week))["n"] >= config.MIN_N:
                trial_week = week
                break
        if trial_week is None and weeks:
            trial_week = weeks[-1]

    results = []
    if trial_week:
        results = analytics.compare_all_groups(
            nitrate("TGID-0", trial_week),
            {g: nitrate(g, trial_week) for g in config.VARIANT_GROUPS})
    control = next((r["control"] for r in results if r["sufficient"]), None)

    weekly = {g: {w: analytics.summarise(nitrate(g, w)) for w in weeks}
              for g in config.TREATMENT_GROUPS}

    growth, yields = {}, {}
    for g in config.TREATMENT_GROUPS:
        routine = [r for r in trials[g]
                   if r["trial_week"] == trial_week and r.get("visit_type") != "HARVEST_VISIT"]
        growth[g] = {"records": len(trials[g]),
                     "height": analytics.summarise([r.get("plant_height_cm") for r in routine]),
                     "leaves": analytics.summarise([r.get("leaf_count") for r in routine])}
        plot = store.get_plot(f"{site_id}-{crop_id}-{g}")
        harvest = [r for r in trials[g] if r.get("visit_type") == "HARVEST_VISIT"]
        yields[g] = analytics.plot_yield(harvest, plot["area_m2"] if plot else None)

    return {
        "site_id": site_id, "crop_id": crop_id,
        "start_date": site["start_date"] if site else None,
        "source": getattr(store, "name", "cloud database"),
        "weeks": weeks, "trial_week": trial_week,
        "total_records": sum(len(rows) for rows in trials.values()),
        "observers": len({r.get("observer_id") for rows in trials.values() for r in rows}),
        "results": results, "control": control, "weekly": weekly,
        "growth": growth, "yields": yields,
        "protocols": {g: store.get_protocol(g) for g in config.TREATMENT_GROUPS},
    }


# --------------------------------------------------------------- helpers ----
def _fmt(value, digits=1) -> str:
    return "-" if value is None else f"{value:.{digits}f}"


def _pct(value) -> str:
    return "-" if value is None else f"{value:+.1f}%"


def _md(text: str) -> str:
    """Tiny Markdown -> reportlab markup (bold and italics only), safely escaped."""
    t = escape(text)
    t = re.sub(r"\*\*(.+?)\*\*", r"<b>\1</b>", t)
    return re.sub(r"(?<!\w)_(.+?)_(?!\w)", r"<i>\1</i>", t)

def _table(rows, widths, st, extra=None):
    data = [[Paragraph(escape(str(c)), st["hdr"]) for c in rows[0]]]
    for row in rows[1:]:
        data.append([c if isinstance(c, Paragraph) else Paragraph(escape(str(c)), st["cell"])
                     for c in row])
    table = Table(data, colWidths=widths, repeatRows=1)
    table.setStyle(TableStyle([
        ("BACKGROUND", (0, 0), (-1, 0), GREEN),
        ("ROWBACKGROUNDS", (0, 1), (-1, -1), [colors.white, LIGHT]),
        ("GRID", (0, 0), (-1, -1), 0.4, colors.HexColor("#c9d1cc")),

        ("TOPPADDING", (0, 0), (-1, -1), 3), ("BOTTOMPADDING", (0, 0), (-1, -1), 3),
        *(extra or [])]))
    return table

def _text(*args, **kwargs):
    """Chart label in Helvetica (reportlab's default would be Times)."""
    kwargs.setdefault("fontName", "Helvetica")

def _ci_chart(control, results, width=470, height=210) -> Drawing:
    """Mean and 95% CI for the control and every group that has enough data."""
    d = Drawing(width, height)
    left, right, bottom, top = 50, 12, 36, 24
    pw, ph = width - left - right, height - bottom - top

    items = [("TGID-0", control, "control", "")]
    for r in results:
        note = "" if r["sufficient"] else f"n = {r['n_variant']} (needs {r['min_n']})"
        items.append((r["group"], r["variant"] if r["sufficient"] else None, r.get("band"), note))

    stats = [i[1] for i in items if i[1]]
    lo, hi = min(s["ci_lower"] for s in stats), max(s["ci_upper"] for s in stats)
    pad = (hi - lo) * 0.18 or 1.0
    lo, hi = lo - pad, hi + pad

    def y(v):
        return bottom + (v - lo) / (hi - lo) * ph

    for k in range(5):                                   # horizontal grid + y labels
        v = lo + (hi - lo) * k / 4
        d.add(Line(left, y(v), left + pw, y(v), strokeColor=GRID, strokeWidth=0.5))
        d.add(_text(left - 6, y(v) - 3, f"{v:.0f}", fontSize=7, textAnchor="end", fillColor=GREY))
    d.add(Line(left, bottom, left + pw, bottom, strokeColor=GREY, strokeWidth=0.8))
    d.add(Line(left, bottom, left, bottom + ph, strokeColor=GREY, strokeWidth=0.8))

    d.add(Rect(left, y(control["ci_lower"]), pw, y(control["ci_upper"]) - y(control["ci_lower"]),
               fillColor=BAND, strokeColor=None))      # the control's 95% band across the chart
    d.add(Line(left, y(control["mean"]), left + pw, y(control["mean"]),
               strokeColor=GREY, strokeWidth=1, strokeDashArray=[4, 3]))

    slot = pw / len(items)
    for i, (group, s, band, note) in enumerate(items):
        cx = left + slot * (i + 0.5)
        d.add(_text(cx, bottom - 13, group, fontSize=7.5, textAnchor="middle", fillColor=DARK))
        if s is None:
            d.add(_text(cx, bottom + ph / 2, "insufficient data", fontSize=6.5,
                         textAnchor="middle", fillColor=GREY))
            d.add(_text(cx, bottom + ph / 2 - 9, note, fontSize=6.5, textAnchor="middle",
                         fillColor=GREY))
            continue
        color = GREY if band == "control" else (GREEN if band == "green" else YELLOW)
        d.add(Line(cx, y(s["ci_lower"]), cx, y(s["ci_upper"]), strokeColor=color, strokeWidth=2.2))
        for edge in (s["ci_lower"], s["ci_upper"]):
            d.add(Line(cx - 6, y(edge), cx + 6, y(edge), strokeColor=color, strokeWidth=2.2))
        d.add(Circle(cx, y(s["mean"]), 4.5, fillColor=color, strokeColor=DARK, strokeWidth=0.8))
        d.add(_text(cx + 9, y(s["mean"]) + 3, f"{s['mean']:.1f}", fontSize=7, fillColor=DARK))
    d.add(_text(left, height - 12, "Nitrate (mg/L): mean and 95% confidence interval",
                 fontSize=8.5, fontName="Helvetica-Bold", fillColor=DARK))
    return d


def _yield_chart(yields: dict, width=470, height=190) -> Drawing:
    d = Drawing(width, height)
    left, right, bottom, top = 50, 12, 30, 24
    pw, ph = width - left - right, height - bottom - top
    top_value = max([v for v in yields.values() if v is not None] or [1]) * 1.18

    def y(v):
        return bottom + v / top_value * ph

    for k in range(5):
        v = top_value * k / 4
        d.add(Line(left, y(v), left + pw, y(v), strokeColor=GRID, strokeWidth=0.5))
        d.add(_text(left - 6, y(v) - 3, f"{v:.0f}", fontSize=7, textAnchor="end", fillColor=GREY))
    d.add(Line(left, bottom, left + pw, bottom, strokeColor=GREY, strokeWidth=0.8))

    slot = pw / len(yields)
    for i, (group, value) in enumerate(yields.items()):
        cx = left + slot * (i + 0.5)
        d.add(_text(cx, bottom - 13, group, fontSize=7.5, textAnchor="middle", fillColor=DARK))
        if value is None:
            d.add(_text(cx, bottom + 6, "in progress", fontSize=7, textAnchor="middle",
                         fillColor=GREY))
            continue
        d.add(Rect(cx - 17, bottom, 34, y(value) - bottom,
                   fillColor=GREY if group == "TGID-0" else GREEN, strokeColor=None))
        d.add(_text(cx, y(value) + 4, f"{value:.0f}", fontSize=7.5, textAnchor="middle",
                     fillColor=DARK))
    d.add(_text(left, height - 12, "Yield (g/m2) per treatment group", fontSize=8.5,
                 fontName="Helvetica-Bold", fillColor=DARK))
    return d

def _findings(data: dict) -> list:
    week, results = data["trial_week"], data["results"]
    if not week:
        return ["No trial records have been captured for this site and crop yet."]
    clear = [r for r in results if r["sufficient"] and not r["overlap"]]
    unclear = [r for r in results if r["sufficient"] and r["overlap"]]
    short = [r for r in results if not r["sufficient"]]

    lines = [f"In {week}, nitrate (mg/L) in the control group (TGID-0) was compared with "
             f"{len(results)} treatment groups using 95% confidence intervals (Student's t)."]
    if clear:
        lines.append("<b>Clear difference</b> from the control (intervals do not overlap): " +
                     "; ".join(f"{r['group']} ({_pct(r['difference_pct'])}, "
                               f"{'higher' if r['difference'] > 0 else 'lower'})" for r in clear)
                     + ".")
    if unclear:
        lines.append("<b>No clear difference</b> yet (intervals overlap): " +
                     "; ".join(f"{r['group']} ({_pct(r['difference_pct'])})" for r in unclear)
                     + ".")
    if short:
        lines.append("<b>Insufficient data</b> (fewer than 5 readings in a group): " +
                     "; ".join(f"{r['group']} (n = {r['n_variant']}, control n = {r['n_control']})"
                               for r in short) + ".")

    done = {g: v for g, v in data["yields"].items() if v is not None}
    if "TGID-0" in done and len(done) > 1:
        best = max(done, key=done.get)
        if best == "TGID-0":
            lines.append(f"Yield: the control had the highest yield ({done['TGID-0']:.0f} g/m2).")
        else:
            gain = (done[best] - done["TGID-0"]) / done["TGID-0"] * 100
            lines.append(f"Yield: {best} was highest at {done[best]:.0f} g/m2 "
                         f"({gain:+.1f}% compared with the control at {done['TGID-0']:.0f} g/m2).")
    else:
        lines.append("Yield comparison: harvests are not yet complete for all groups.")
    return lines


def _protocol_blocks(protocols: dict):
    """(description per group, shared trial conditions taken from the control protocol)."""
    descriptions = {}
    for group, text in protocols.items():
        descriptions[group] = _md(text.split("\n\n")[0]) if text else \
            "<i>Protocol text not yet confirmed by the research team.</i>"
    control = protocols.get("TGID-0") or ""
    shared = []
    if "\n\n" in control:
        for line in control.split("\n\n", 1)[1].splitlines():
            if line.strip():
                shared.append(_md(line.strip().lstrip("- ").strip()))
    return descriptions, shared

"""Password hashing helpers (salted PBKDF2). Passwords are never stored in plain text."""
import hashlib
import hmac
import os

_ITERATIONS = 200_000


def hash_password(password: str, salt_hex: str = None):
    """Return (salt_hex, hash_hex). A new random salt is made when none is given."""
    salt = bytes.fromhex(salt_hex) if salt_hex else os.urandom(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, _ITERATIONS)
    return salt.hex(), digest.hex()


def verify_password(password: str, salt_hex: str, hash_hex: str) -> bool:
    """Constant-time check of a password against a stored salt and hash."""
    _, candidate = hash_password(password, salt_hex)
    return hmac.compare_digest(candidate, hash_hex)

def build_nitrate() -> dict:
    """(group, week) -> list of nitrate readings, for TW-1..TW-6 and all 6 groups.

    The random numbers are seeded per group and week, so the data is identical every run.

    data = {}

            data[(tgid, week)] = [round(rng.gauss(mean, NITRATE_SD), 1) for _ in range(count)]
    return data

"""Offline-first sync engine (Module 5).

Records are always saved on the phone first as PENDING. When the phone is
online, this worker pushes them (oldest first) to the FastAPI engine, which
validates and writes to Supabase:

    200 SUCCESS  -> SYNCED, cloud Record ID (EP-T###) stored locally
    422 REJECTED -> FLAGGED_FOR_REVIEW with the reason (farmer can fix it)
    anything else (timeout, 5xx, no connection) -> stays PENDING, retried later

`post` and `get` can be swapped for a test client so the whole path can be
tested without a running server.
"""
import threading
import time

import requests

import config
import database as db

HEADERS = {"X-User-Role": config.ROLE_FARMER}


def is_online(db_path: str = None, api_url: str = None, get=requests.get) -> bool:
    """Online = demo offline switch is OFF, the API answers, and the cloud is reachable."""
    if db.is_offline_mode(db_path):
        return False
    try:
        resp = get(f"{api_url if api_url is not None else config.API_URL}/health", timeout=3)
        return resp.status_code == 200 and bool(resp.json().get("cloud"))
    except Exception:
        return False


def sync_setup_data(db_path: str = None, api_url: str = None, post=requests.post,
                    get=requests.get) -> None:
    """Push local site/plot setup, then pull the cloud's version ("Server Wins")."""
    base = api_url if api_url is not None else config.API_URL
    for site in db.list_sites(db_path):
        resp = post(f"{base}/api/v1/sites", json=site, headers=HEADERS, timeout=5)
        if resp.status_code == 200:
            db.upsert_site(resp.json()["site_id"], resp.json()["start_date"], db_path)
    for plot in db.list_plots(db_path):
        resp = post(f"{base}/api/v1/plots", json=plot, headers=HEADERS, timeout=5)
        if resp.status_code == 200:
            p = resp.json()
            db.upsert_plot(p["plot_id"], p["site_id"], p["crop_id"],
                           p["treatment_group_id"], p["area_m2"], db_path)
    # Pull anything the cloud already has (e.g. set up by another farmer)
    any_role = {"X-User-Role": config.ROLE_FARMER}
    for site in get(f"{base}/api/v1/sites", headers=any_role, timeout=5).json():
        db.upsert_site(site["site_id"], site["start_date"], db_path)
    for p in get(f"{base}/api/v1/plots", headers=any_role, timeout=5).json():
        db.upsert_plot(p["plot_id"], p["site_id"], p["crop_id"],
                       p["treatment_group_id"], p["area_m2"], db_path)


def sync_pending_records(db_path: str = None, api_url: str = None, post=requests.post,
                         get=requests.get) -> dict:
    """Push every PENDING record, oldest first. Returns a summary dict."""
    base = api_url if api_url is not None else config.API_URL
    pending = db.list_records(db_path, status="PENDING", oldest_first=True)
    summary = {"online": True, "synced": 0, "flagged": 0, "pending_left": 0,
               "message": ""}

    if not is_online(db_path, base, get):
        summary.update(online=False, pending_left=len(pending),
                       message=f"Offline mode: {len(pending)} record(s) stored locally. "
                               f"Awaiting internet connection.")
        return summary

    try:
        sync_setup_data(db_path, base, post, get)
    except Exception:
        pass  # setup sync is best effort; records can still be pushed

    for rec in pending:
        payload = {c: rec[c] for c in db.PAYLOAD_COLUMNS}
        payload["is_harvest_complete"] = bool(rec["is_harvest_complete"])
        try:
            resp = post(f"{base}/api/v1/trials", json=payload, headers=HEADERS, timeout=8)
        except Exception:
            break  # connection dropped: leave the rest PENDING and retry later
        if resp.status_code == 200 and resp.json().get("status") == "SUCCESS":
            db.mark_synced(rec["local_id"], resp.json()["cloud_record_id"], db_path)
            summary["synced"] += 1
        elif resp.status_code == 422:
            db.mark_flagged(rec["local_id"], resp.json().get("error", "Rejected by cloud"), db_path)
            summary["flagged"] += 1
        # any other status (e.g. 503): stays PENDING

    summary["pending_left"] = db.count_by_status(db_path)["PENDING"]
    summary["message"] = (f"Cloud sync completed: {summary['synced']} synced, "
                          f"{summary['flagged']} flagged, {summary['pending_left']} pending.")
    return summary


def start_background_sync(db_path: str = None, api_url: str = None, interval: int = 15):
    """Daemon thread: whenever records are PENDING and the phone is online, sync them."""
    def loop():
        while True:
            try:
                if db.count_by_status(db_path)["PENDING"] > 0:
                    sync_pending_records(db_path, api_url)
            except Exception:
                pass
            time.sleep(interval)

    thread = threading.Thread(target=loop, daemon=True, name="agritrial-sync")
    thread.start()
    return thread

"""Unit tests for the confidence-interval engine and yield calculation."""
import math

import numpy as np
import pytest
from scipy import stats

from analytics import compare_nitrate, mean_ci, plot_yield


def test_mean_ci_matches_hand_calculation():
    data = [10, 12, 11, 13, 14]
    r = mean_ci(data)
    sd = np.std(data, ddof=1)
    margin = stats.t.ppf(0.975, 4) * sd / math.sqrt(5)
    assert r["mean"] == pytest.approx(12.0)
    assert r["margin"] == pytest.approx(margin)
    assert r["ci_lower"] == pytest.approx(12.0 - margin)
    assert r["ci_upper"] == pytest.approx(12.0 + margin)
    assert r["t_value"] == pytest.approx(2.776, abs=1e-3)  # t(0.975, df=4)


def test_fewer_than_five_blocks_the_chart():
    result = compare_nitrate([1, 2, 3, 4, 5], [1, 2, 3, 4])
    assert result["sufficient"] is False
    assert "More records needed" in result["notice"]
    assert (result["n_control"], result["n_variant"]) == (5, 4)


def test_blank_nitrate_values_do_not_count_towards_n():
    result = compare_nitrate([1, 2, None, 4, 5, None], [1, 2, 3, 4, 5])
    assert result["n_control"] == 4 and result["sufficient"] is False


def test_five_in_both_groups_gives_ci():
    result = compare_nitrate([110, 112, 108, 111, 109], [140, 142, 138, 141, 139])
    assert result["sufficient"] is True
    assert result["band"] == "green" and result["overlap"] is False


def test_overlapping_intervals_are_yellow():
    result = compare_nitrate([100, 120, 90, 130, 110], [105, 125, 95, 128, 112])
    assert result["sufficient"] is True and result["band"] == "yellow"


def test_yield_is_total_fresh_weight_over_area():

"""Shared validation rules (Milestone 1, Section 6b).

The SAME function runs on the phone (before saving) and in the API (before
writing to the cloud), so a record that passes on the phone is judged by the
same rules as the one the cloud applies.
"""
import math
import re
from datetime import date

from config import TREATMENT_GROUPS, VISIT_TYPES

# The 11 mandatory fields (Milestone 1, Section 2c).
MANDATORY_FIELDS = (
    "local_id",            # Record ID (temporary on the device)
    "entry_date",
    "site_id",
    "environment_id",
    "crop_id",
    "treatment_group_id",
    "visit_type",
    "trial_week",
    "observer_id",
    "plant_height_cm",
    "leaf_count",
)

# field: (low, high, low_inclusive, high_inclusive, label)
RANGE_RULES = {
    "plant_height_cm": (0.0, 500.0, False, False, "Plant height (cm)"),
    "leaf_count": (0, 500, True, False, "Leaf count"),
    "fresh_weight_g": (0.0, 10000.0, False, False, "Fresh weight (g)"),
    "dry_weight_g": (0.0, 10000.0, False, False, "Dry weight (g)"),
    "spad_index": (0.0, 100.0, True, True, "SPAD index"),
    "root_length_cm": (0.0, 200.0, False, False, "Root length (cm)"),
    "ph_level": (0.0, 14.0, True, True, "pH"),
    "ec_level": (0.0, 10.0, True, True, "EC (mS/cm)"),
    "dissolved_oxygen": (0.0, 20.0, True, True, "Dissolved O2 (mg/L)"),
    "temperature_c": (-10.0, 60.0, True, True, "Temperature (C)"),
    "humidity_pct": (0.0, 100.0, True, True, "Humidity (%)"),
    "nitrate_mg_l": (0.0, 1000.0, False, False, "Nitrate (mg/L)"),
    "phosphate_mg_l": (0.0, 500.0, False, False, "Phosphate (mg/L)"),
}

_TRIAL_WEEK = re.compile(r"^TW-[1-9][0-9]*$")
_OBSERVER = re.compile(r"^OID-[0-9]{3}$")


def is_blank(value) -> bool:
    """True for None, empty/whitespace strings and NaN."""
    if value is None:
        return True
    if isinstance(value, str) and not value.strip():
        return True
    if isinstance(value, float) and math.isnan(value):
        return True
    return False


def _to_number(value) -> float:
    """Convert to a finite float or raise ValueError."""
    if isinstance(value, bool):
        raise ValueError("must be a number")
    try:
        number = float(value)
    except (TypeError, ValueError):
        raise ValueError("must be a number")
    if math.isnan(number) or math.isinf(number):
        raise ValueError("must be a number")
    return number


def _range_text(low, high, low_inc, high_inc) -> str:
    low_word = "at least" if low_inc else "more than"
    high_word = "at most" if high_inc else "less than"
    return f"must be {low_word} {low} and {high_word} {high}"


def validate_record(rec: dict) -> list:
    """Validate one trial record.

    Returns a list of (field, message) tuples. An empty list means the record
    is valid. Optional fields may be blank; if filled they must be in range.
    """
    errors = []

    # 1. Mandatory fields present
    for field in MANDATORY_FIELDS:
        if is_blank(rec.get(field)):
            errors.append((field, "is mandatory"))

    # 2. Format checks on the identifier-style fields
    iso = rec.get("entry_date")
    if not is_blank(iso):
        try:
            date.fromisoformat(str(iso))
        except ValueError:
            errors.append(("entry_date", "must be a valid date (YYYY-MM-DD)"))
    if not is_blank(rec.get("visit_type")) and rec["visit_type"] not in VISIT_TYPES:
        errors.append(("visit_type", "must be ROUTINE_MONITORING or HARVEST_VISIT"))
    if (not is_blank(rec.get("treatment_group_id"))
            and rec["treatment_group_id"] not in TREATMENT_GROUPS):
        errors.append(("treatment_group_id", "must be TGID-0 to TGID-5"))
    if not is_blank(rec.get("trial_week")) and not _TRIAL_WEEK.match(str(rec["trial_week"])):
        errors.append(("trial_week", "must look like TW-1, TW-2, ..."))
    if not is_blank(rec.get("observer_id")) and not _OBSERVER.match(str(rec["observer_id"])):
        errors.append(("observer_id", "must look like OID-###"))

    # 3. Numeric ranges (mandatory metrics are already reported as missing above)
    numbers = {}
    for field, (low, high, low_inc, high_inc, label) in RANGE_RULES.items():
        value = rec.get(field)
        if is_blank(value):
            continue
        try:
            number = _to_number(value)
        except ValueError as exc:
            errors.append((field, f"{label} {exc}"))
            continue
        if field == "leaf_count" and not number.is_integer():
            errors.append((field, "Leaf count must be a whole number"))
            continue
        ok_low = number >= low if low_inc else number > low
        ok_high = number <= high if high_inc else number < high
        if not (ok_low and ok_high):
            errors.append((field, f"{label} {_range_text(low, high, low_inc, high_inc)}"))
        else:
            numbers[field] = number

    # 4. Null-safe cross-field rule: dry weight <= fresh weight (only if both filled)
    if "dry_weight_g" in numbers and "fresh_weight_g" in numbers:
        if numbers["dry_weight_g"] > numbers["fresh_weight_g"]:
            errors.append(("dry_weight_g", "Dry weight cannot be more than fresh weight"))

    return errors


def format_errors(errors: list) -> str:
    """Turn the error list into one readable sentence for the UI / API."""
    return "; ".join(f"{field}: {message}" for field, message in errors)
